READY TO INTEGRATE

One map. Your stack.

Start in plain HTML, then reuse the same helpers in React, Next.js, or any web app.

● ● ● HTML · 6 lines 
<div id="map" style="height:420px"></div>
<script type="module">
  import * as ml from "https://unpkg.com/maplibre-gl@6.16.0/dist/maplibre-gl.mjs";
  import { HarshMaps } from "https://cdn.jsdelivr.net/gh/harshyadavv2456/HarshMaps@main/sdk/harshmaps.js";
  HarshMaps.createMap(ml, "map", { style: HarshMaps.styles.liberty });
</script>
